# core/activity_engine.py
import sys
import time
import threading
import os
from PyQt6.QtCore import QObject, pyqtSignal
from core.config_manager import config_mgr
from core.window_tracker import get_active_window_info
from core.input_engine import get_last_activity_time, mark_activity

class ActivityEngine(QObject):
    tick = pyqtSignal(dict)
    active_window_changed = pyqtSignal(str)

    def __init__(self, parent=None):
        super().__init__(parent)
        self.current_window_id = None
        self.current_window_title = ""
        self.current_app_name = ""
        self.running = False
        self.my_pid = os.getpid()
        self.worker_thread = None
        self._last_disk_save_time = time.time()

    def start(self):
        if self.running: return
        self.running = True
        self.worker_thread = threading.Thread(target=self._engine_loop, daemon=True)
        self.worker_thread.start()

    def _engine_loop(self):
        last_eval_time = time.time()
        last_window_poll_time = 0.0
        poll_interval = 0.7 if sys.platform == 'darwin' else 0.15

        while self.running:
            now = time.time()
            if now - last_window_poll_time >= poll_interval:
                last_window_poll_time = now
                try:
                    win_info = get_active_window_info()
                    if win_info and win_info.pid != self.my_pid:
                        if self.current_window_id != win_info.window_id:
                            self.current_window_id = win_info.window_id
                            self.current_window_title = win_info.title
                            self.current_app_name = win_info.app_name
                            mark_activity()
                            self.active_window_changed.emit(win_info.format_slot_string())
                        else:
                            self.current_window_title = win_info.title
                            if win_info.app_name:
                                self.current_app_name = win_info.app_name
                except Exception:
                    pass

            if now - last_eval_time >= 1.0:
                last_eval_time = now
                self._evaluate_timers(now)

            time.sleep(0.05)

    @staticmethod
    def _clean_app_token(app_str: str) -> str:
        s = (app_str or "").strip().lower()
        for ext in (".exe", ".app", ".bin"):
            if s.endswith(ext):
                s = s[:-len(ext)].strip()
        return s

    def _is_window_matched_for_timer(self, timer_data: dict, window_title: str, app_name: str) -> bool:
        group_name = timer_data.get("group", "")
        if not group_name: return False
        groups = config_mgr.config.get("groups", {})
        target_keywords = groups.get(group_name, [])
        if not target_keywords: return False

        cur_title = (window_title or "").strip().lower()
        cur_app = (app_name or "").strip().lower()
        clean_cur_app = self._clean_app_token(cur_app)
        if not cur_title and not cur_app: return False

        for kw in target_keywords:
            if not kw or not isinstance(kw, str): continue
            raw_kw = kw.strip()
            if not raw_kw: continue

            if raw_kw.startswith("[") and "]" in raw_kw:
                close_idx = raw_kw.find("]")
                slot_app = raw_kw[1:close_idx].strip().lower()
                slot_title = raw_kw[close_idx + 1:].strip().lower()
                clean_slot_app = self._clean_app_token(slot_app)
                if clean_slot_app and clean_cur_app and (clean_slot_app == clean_cur_app or slot_app == cur_app):
                    return True
                if slot_title and cur_title and (slot_title in cur_title or cur_title in slot_title):
                    return True

            clean_kw = raw_kw.lower()
            clean_kw_app = self._clean_app_token(clean_kw)
            if cur_title and clean_kw in cur_title: return True
            if cur_app and (clean_kw in cur_app or (clean_kw_app and clean_kw_app == clean_cur_app)): return True
            if len(clean_cur_app) >= 3 and clean_cur_app in clean_kw: return True

        return False

    def is_timer_active(self, timer_id: str) -> bool:
        if not timer_id: return True
        timers = config_mgr.config.get("timers", [])
        target_timer = next((t for t in timers if t.get("id") == timer_id), None)
        if not target_timer: return True
        if target_timer.get("paused", False): return False
        timer_timeout = target_timer.get("idle_timeout", 5)
        if (time.time() - get_last_activity_time()) > timer_timeout: return False
        return self._is_window_matched_for_timer(target_timer, self.current_window_title, self.current_app_name)

    def _evaluate_timers(self, now):
        active_title = self.current_window_title
        app_name = self.current_app_name
        input_elapsed = now - get_last_activity_time()
        timers = config_mgr.config.get("timers", [])
        updated = {}

        for t in timers:
            if t.get("paused", False): continue
            timer_timeout = t.get("idle_timeout", 5)
            if input_elapsed > timer_timeout: continue
            if self._is_window_matched_for_timer(t, active_title, app_name):
                t["elapsed_seconds"] = t.get("elapsed_seconds", 0) + 1
                updated[t["id"]] = t["elapsed_seconds"]

        if updated:
            self.tick.emit(updated)
            if now - self._last_disk_save_time >= 30.0:
                self._last_disk_save_time = now
                config_mgr.save_config()

activity_engine = ActivityEngine()

# core/config_manager.py
import os
import sys
import json
import uuid
import shutil
import threading

def get_bundle_dir():
    if getattr(sys, 'frozen', False):
        if hasattr(sys, '_MEIPASS'):
            return sys._MEIPASS
        exe_dir = os.path.dirname(sys.executable)
        internal_dir = os.path.join(exe_dir, "_internal")
        if os.path.exists(internal_dir):
            return internal_dir
        if sys.platform == 'darwin':
            res_dir = os.path.join(os.path.dirname(exe_dir), "Resources")
            if os.path.exists(res_dir):
                return res_dir
        return exe_dir
    else:
        return os.path.dirname(os.path.dirname(os.path.abspath(__file__)))

BUNDLE_DIR = get_bundle_dir()

if sys.platform == 'darwin':
    USER_DATA_DIR = os.path.expanduser("~/Library/Application Support/morningmeal_ComPet")
else:
    USER_DATA_DIR = os.path.join(os.path.expanduser("~"), ".morningmeal_ComPet")

CONFIG_PATH = os.path.join(USER_DATA_DIR, "config.json")
SKINS_DIR = os.path.join(USER_DATA_DIR, "skins")
USER_SOUNDS_DIR = os.path.join(USER_DATA_DIR, "sounds")
CRASH_LOG_PATH = os.path.join(USER_DATA_DIR, "crash_log.txt")

BUNDLE_SKINS_DIR = os.path.join(BUNDLE_DIR, "assets", "skins")
BUNDLE_SOUNDS_DIR = os.path.join(BUNDLE_DIR, "assets", "sounds")

DEFAULT_CONFIG = {
    "settings": {
        "language": "auto",
        "dark_mode": False,
        "tray_mode": False,
        "click_through": False,
        "clamp_to_screen": True,
        "magnetic_snap": True,

            "idle_timeout": 5,
            "elapsed_seconds": 0,
            "x": 200,
            "y": 300,
            "paused": False
        }
    ],
    "pets": []
}

class ConfigManager:
    def __init__(self):
        self._lock = threading.RLock()
        self._ensure_dirs()
        self.config = self.load_config()

    def _ensure_dirs(self):
        os.makedirs(USER_DATA_DIR, exist_ok=True)
        os.makedirs(SKINS_DIR, exist_ok=True)
        os.makedirs(USER_SOUNDS_DIR, exist_ok=True)

        default_user_skin = os.path.join(SKINS_DIR, "default")
        bundle_default_skin = os.path.join(BUNDLE_SKINS_DIR, "default")

        if not os.path.exists(default_user_skin):
            if os.path.exists(bundle_default_skin):
                shutil.copytree(bundle_default_skin, default_user_skin)
            else:
                os.makedirs(default_user_skin, exist_ok=True)
                self.save_skin_config("default", {
                    "name": "default",
                    "squash_depth": 0.20,
                    "enable_bounce": True,
                    "idle_image": "idle.png",
                    "tap_images": ["tap_left.png", "tap_right.png"],
                    "key_mappings": {}
                })

    def load_config(self):
        with self._lock:
            data = json.loads(json.dumps(DEFAULT_CONFIG))
            if os.path.exists(CONFIG_PATH):
                try:
                    with open(CONFIG_PATH, "r", encoding="utf-8") as f:
                        loaded = json.load(f)
                        for k, v in loaded.items():
                            if isinstance(v, dict) and k in data:
                                data[k].update(v)
                            else:
                                data[k] = v
                except Exception as e:
                    print(f"[ConfigManager] Load error: {e}")

            if not data.get("pets"):
                first_timer_id = data["timers"][0]["id"] if data.get("timers") else ""
                data["pets"].append({
                    "id": str(uuid.uuid4()),
                    "skin": "default",
                    "scale": 1.0,
                    "x": 200,
                    "y": 200,
                    "bound_timer_id": first_timer_id
                })
            return data

    def save_config(self):
        with self._lock:
            tmp_path = CONFIG_PATH + ".tmp"
            try:
                with open(tmp_path, "w", encoding="utf-8") as f:
                    json.dump(self.config, f, indent=4, ensure_ascii=False)
                os.replace(tmp_path, CONFIG_PATH)
            except Exception as e:
                print(f"[ConfigManager] Save error: {e}")
                if os.path.exists(tmp_path):
                    try:
                        os.remove(tmp_path)
                    except Exception:
                        pass

    def get_skin_config(self, skin_name):
        with self._lock:
            path = os.path.join(SKINS_DIR, skin_name, "config.json")
            conf = {
                "name": skin_name,
                "squash_depth": 0.20,
                "enable_bounce": True,
                "idle_image": "idle.png",
                "tap_images": ["tap_left.png", "tap_right.png"],
                "key_mappings": {}
            }
            if os.path.exists(path):
                try:
                    with open(path, "r", encoding="utf-8") as f:
                        conf.update(json.load(f))
                except Exception:
                    pass
            return conf

    def save_skin_config(self, skin_name, data):
        with self._lock:
            target_dir = os.path.join(SKINS_DIR, skin_name)
            os.makedirs(target_dir, exist_ok=True)
            path = os.path.join(target_dir, "config.json")
            tmp_path = path + ".tmp"
            try:
                with open(tmp_path, "w", encoding="utf-8") as f:
                    json.dump(data, f, indent=4, ensure_ascii=False)
                os.replace(tmp_path, path)
            except Exception as e:
                print(f"[ConfigManager] Skin config save error for {skin_name}: {e}")

config_mgr = ConfigManager()

class I18nManager(QObject):
    language_changed = pyqtSignal()

    def __init__(self):
        super().__init__()
        self._current_lang = "ko"

    def detect_os_language(self):
        sys_lang = QLocale.system().name()
        if sys_lang.startswith("en"):
            self.set_language("en")
        elif sys_lang.startswith("ja"):
            self.set_language("ja")
        elif sys_lang == "zh_CN" or sys_lang.startswith("zh_Hans"):
            self.set_language("zh_CN")
        elif sys_lang.startswith("zh"):
            self.set_language("zh_TW")
        else:
            self.set_language("ko")

    def set_language(self, lang_code):
        if lang_code in TRANSLATIONS and self._current_lang != lang_code:
            self._current_lang = lang_code
            self.language_changed.emit()

    def get_lang(self):
        return self._current_lang

    def tr(self, key, default=None):
        fallback = default if default is not None else key
        return TRANSLATIONS.get(self._current_lang, TRANSLATIONS["ko"]).get(key, fallback)

I18n = I18nManager()

# core/input_engine.py
import sys
import time
import threading
import ctypes
import subprocess
from pynput import keyboard, mouse
from PyQt6.QtCore import QObject, pyqtSignal

class InputEngineBridge(QObject):
    input_occurred = pyqtSignal(str, bool)
    mouse_clicked_global = pyqtSignal(str, int, int)  # 버튼명, x, y 좌표 포함
    escape_pressed_global = pyqtSignal()

input_bridge = InputEngineBridge()

active_modifiers = set()
last_activity_time = time.time()
_last_move_throttle_time = 0.0

SHIFT_MAP = {
    '1': '!', '2': '@', '3': '#', '4': '$', '5': '%',
    '6': '^', '7': '&', '8': '*', '9': '(', '0': ')',
    '-': '_', '=': '+', '[': '{', ']': '}', '\\': '|',
    ';': ':', "'": '"', ',': '<', '.': '>', '/': '?', '`': '~'
}

MAC_VK_MAP = {
    0: 'a', 1: 's', 2: 'd', 3: 'f', 4: 'h', 5: 'g', 6: 'z', 7: 'x',
    8: 'c', 9: 'v', 11: 'b', 12: 'q', 13: 'w', 14: 'e', 15: 'r',
    16: 'y', 17: 't', 18: '1', 19: '2', 20: '3', 21: '4', 22: '6',
    23: '5', 24: '=', 25: '9', 26: '7', 27: '-', 28: '8', 29: '0',
    30: ']', 31: 'o', 32: 'u', 33: '[', 34: 'i', 35: 'p', 37: 'l',
    38: 'j', 39: "'", 40: 'k', 41: ';', 42: '\\', 43: ',', 44: '/',
    45: 'n', 46: 'm', 47: '.', 49: 'space', 50: '`'
}

def get_last_activity_time() -> float:
    return last_activity_time

def mark_activity():
    global last_activity_time
    last_activity_time = time.time()

def check_mac_accessibility():
    if sys.platform != 'darwin':
        return True
    try:
        app_services = ctypes.cdll.LoadLibrary('/System/Library/Frameworks/ApplicationServices.framework/ApplicationServices')
        app_services.AXIsProcessTrusted.restype = ctypes.c_bool
        is_trusted = app_services.AXIsProcessTrusted()
        if not is_trusted:
            subprocess.Popen(
                ["open", "x-apple.systempreferences:com.apple.preference.security?Privacy_Accessibility"],
                stderr=subprocess.DEVNULL
            )
        return bool(is_trusted)
    except Exception as e:
        print(f"[InputEngine] Accessibility check failed: {e}")
        return False

def normalize_key_token(token):
    if not token:
        return ""
    token_str = str(token).strip()
    if token_str.startswith("<") and token_str.endswith(">"):
        inner = token_str[1:-1]
        if inner.isdigit():
            vk = int(inner)
            if 48 <= vk <= 57:
                return chr(vk)
            elif 65 <= vk <= 90:
                return chr(vk).lower()
            elif 96 <= vk <= 105:
                return str(vk - 96)
    return token_str.lower()

def _extract_vk_fallback(key) -> str:
    vk = getattr(key, 'vk', None)
    if vk is None:
        return ""
    if sys.platform == 'darwin':
        return MAC_VK_MAP.get(vk, "")
    else:
        if 48 <= vk <= 57:
            return chr(vk)
        elif 65 <= vk <= 90:
            return chr(vk).lower()
        elif 96 <= vk <= 105:
            return str(vk - 96)
    return ""

def on_key_press(key):
    try:
        mark_activity()
        if key in (keyboard.Key.shift, keyboard.Key.shift_r, keyboard.Key.shift_l):
            active_modifiers.add("shift")
            return
        elif key in (keyboard.Key.ctrl, keyboard.Key.ctrl_l, keyboard.Key.ctrl_r):
            active_modifiers.add("ctrl")
            return
        elif key in (keyboard.Key.alt, keyboard.Key.alt_l, keyboard.Key.alt_r, keyboard.Key.alt_gr):
            active_modifiers.add("alt")
            return
        elif key in (keyboard.Key.cmd, keyboard.Key.cmd_l, keyboard.Key.cmd_r):
            active_modifiers.add("cmd")
            return

        is_shift = "shift" in active_modifiers
        is_ctrl = "ctrl" in active_modifiers
        is_alt = "alt" in active_modifiers
        is_cmd = "cmd" in active_modifiers

        char_val = getattr(key, 'char', None)
        base_name = None
        

        if char_val is not None:
            raw_char = str(char_val)
            if len(raw_char) == 1 and ord(raw_char) < 32:
                base_name = chr(ord(raw_char) + 96).lower()
            elif len(raw_char) == 1 and ord(raw_char) <= 126:
                base_name = raw_char.lower()
            else:
                base_name = _extract_vk_fallback(key) or raw_char.lower()
        else:
            vk_name = _extract_vk_fallback(key)
            if vk_name:
                base_name = vk_name
            else:
                raw_name = getattr(key, 'name', str(key))
                clean_name = str(raw_name).replace("Key.", "").replace("key.", "")
                base_name = normalize_key_token(clean_name)

        resolved_keys = []
        if char_val and char_val in "!@#$%^&*()_+{}|:\"<>?~":
            resolved_keys.append(char_val)
        if is_shift and base_name in SHIFT_MAP:
            resolved_keys.append(SHIFT_MAP[base_name])

        mods = []
        if is_cmd: mods.append("cmd")
        if is_ctrl: mods.append("ctrl")
        if is_alt: mods.append("alt")
        if is_shift and not (is_shift and base_name in SHIFT_MAP):
            mods.append("shift")

        if mods and base_name:
            resolved_keys.append("+".join(mods) + "+" + base_name)
        if base_name:
            resolved_keys.append(base_name)

        if base_name == "esc":
            input_bridge.escape_pressed_global.emit()

        payload = "|".join(dict.fromkeys(resolved_keys))
        input_bridge.input_occurred.emit(payload, False)
    except Exception as e:
        print(f"[InputEngine] Key error: {e}")

def on_key_release(key):
    try:
        if key in (keyboard.Key.shift, keyboard.Key.shift_r, keyboard.Key.shift_l):
            active_modifiers.discard("shift")
        elif key in (keyboard.Key.ctrl, keyboard.Key.ctrl_l, keyboard.Key.ctrl_r):
            active_modifiers.discard("ctrl")
        elif key in (keyboard.Key.alt, keyboard.Key.alt_l, keyboard.Key.alt_r, keyboard.Key.alt_gr):
            active_modifiers.discard("alt")
        elif key in (keyboard.Key.cmd, keyboard.Key.cmd_l, keyboard.Key.cmd_r):
            active_modifiers.discard("cmd")
    except Exception:
        pass

def on_mouse_click(x, y, button, pressed):
    if button == mouse.Button.left:
        b_name = "mouse_left"
    elif button == mouse.Button.right:
        b_name = "mouse_right"
    elif button == mouse.Button.middle:
        b_name = "mouse_middle"
    else:
        b_name = "mouse_click"

    if pressed:
        mark_activity()
        input_bridge.input_occurred.emit(b_name, True)
    else:
        input_bridge.mouse_clicked_global.emit(b_name, int(x), int(y))

def on_mouse_move(x, y):
    global _last_move_throttle_time
    now = time.time()
    if now - _last_move_throttle_time > 0.15:
        _last_move_throttle_time = now
        mark_activity()

def on_mouse_scroll(x, y, dx, dy):
    mark_activity()

def start_input_engine():
    if sys.platform == 'darwin':
        check_mac_accessibility()

    def listener_thread():
        if sys.platform == 'darwin':
            time.sleep(0.3)
        try:
            with keyboard.Listener(on_press=on_key_press, on_release=on_key_release) as k_listener, \
                 mouse.Listener(on_click=on_mouse_click, on_move=on_mouse_move, on_scroll=on_mouse_scroll) as m_listener:
                k_listener.join()
                m_listener.join()
        except Exception as e:
            print(f"[InputEngine] Listener error: {e}")

    t = threading.Thread(target=listener_thread, daemon=True)
    t.start()

# core/snap_manager.py
from PyQt6.QtCore import QPoint
from core.config_manager import config_mgr

class SnapManager:
    @staticmethod
    def calculate_snap(moving_widget, all_widgets, new_pos: QPoint) -> QPoint:
        if not config_mgr.config.get("settings", {}).get("magnetic_snap", True):
            return new_pos

        threshold = config_mgr.config.get("settings", {}).get("snap_distance", 15)
        target_x = new_pos.x()
        target_y = new_pos.y()
        w = moving_widget.width()
        h = moving_widget.height()

        m_left = target_x
        m_right = target_x + w
        m_top = target_y
        m_bottom = target_y + h

        for other in all_widgets:
            if other == moving_widget or not other.isVisible():
                continue
            o_geo = other.geometry()
            o_left, o_right, o_top, o_bottom = o_geo.left(), o_geo.right(), o_geo.top(), o_geo.bottom()

            if abs(m_left - o_right) <= threshold: target_x = o_right
            elif abs(m_right - o_left) <= threshold: target_x = o_left - w
            elif abs(m_left - o_left) <= threshold: target_x = o_left
            elif abs(m_right - o_right) <= threshold: target_x = o_right - w

            if abs(m_top - o_bottom) <= threshold: target_y = o_bottom
            elif abs(m_bottom - o_top) <= threshold: target_y = o_top - h
            elif abs(m_top - o_top) <= threshold: target_y = o_top
            elif abs(m_bottom - o_bottom) <= threshold: target_y = o_bottom - h

        return QPoint(target_x, target_y)

# core/sound_manager.py
import os
import wave
from PyQt6.QtCore import QUrl
from PyQt6.QtMultimedia import QSoundEffect
from core.config_manager import config_mgr, USER_SOUNDS_DIR, BUNDLE_SOUNDS_DIR

SUPPORTED_EXTENSIONS = ('.wav',)

def scan_sound_files():
    found_sounds = {}
    if os.path.exists(BUNDLE_SOUNDS_DIR):
        for root, _, files in os.walk(BUNDLE_SOUNDS_DIR):
            for f in sorted(files):
                if f.lower().endswith(SUPPORTED_EXTENSIONS):
                    base_name = os.path.splitext(f)[0]
                    full_path = os.path.abspath(os.path.join(root, f))
                    rel_dir = os.path.relpath(root, BUNDLE_SOUNDS_DIR)
                    name = f"[{rel_dir}] {base_name}" if rel_dir != "." else base_name
                    found_sounds[name] = full_path
    if os.path.exists(USER_SOUNDS_DIR):
        for root, _, files in os.walk(USER_SOUNDS_DIR):
            for f in sorted(files):
                if f.lower().endswith(SUPPORTED_EXTENSIONS):
                    base_name = os.path.splitext(f)[0]
                    full_path = os.path.abspath(os.path.join(root, f))
                    found_sounds[f"[User] {base_name}"] = full_path
    return found_sounds

class SoundManager:
    def __init__(self):
        self.key_pool = []
        self.click_pool = []
        self.key_idx = 0
        self.click_idx = 0
        self._initialized = False

    def ensure_initialized(self):
        if self._initialized: return
        self.key_pool = [QSoundEffect() for _ in range(6)]
        self.click_pool = [QSoundEffect() for _ in range(4)]
        self._initialized = True
        self.load_sounds()

    def get_actual_path(self, prefix):
        snd_conf = config_mgr.config.get("sound", {})
        val = snd_conf.get(f"{prefix}_sound_path", "")
        if not val or val == "none": return ""
        if os.path.isabs(val) and os.path.exists(val) and val.lower().endswith('.wav'): return val
        candidate = os.path.join(USER_SOUNDS_DIR, val)
        if os.path.exists(candidate) and candidate.lower().endswith('.wav'): return os.path.abspath(candidate)
        candidate_bundle = os.path.join(BUNDLE_SOUNDS_DIR, val)
        if os.path.exists(candidate_bundle) and candidate_bundle.lower().endswith('.wav'): return os.path.abspath(candidate_bundle)
        scanned = scan_sound_files()
        if val in scanned: return scanned[val]
        return ""

    def load_sounds(self):
        if not self._initialized: return
        k_path = self.get_actual_path("key")
        c_path = self.get_actual_path("click")
        for player in self.key_pool:
            player.setSource(QUrl.fromLocalFile(k_path) if k_path and os.path.exists(k_path) else QUrl())
        for player in self.click_pool:
            player.setSource(QUrl.fromLocalFile(c_path) if c_path and os.path.exists(c_path) else QUrl())
        self.update_volumes()

    def update_volumes(self):
        if not self._initialized: return
        snd_conf = config_mgr.config.get("sound", {})
        k_vol = snd_conf.get("key_volume", 50) / 100.0
        c_vol = snd_conf.get("click_volume", 50) / 100.0
        for p in self.key_pool: p.setVolume(k_vol)
        for p in self.click_pool: p.setVolume(c_vol)

    def play_key(self):
        if not self._initialized or not self.key_pool: return
        if not config_mgr.config.get("sound", {}).get("key_sound_enabled", True): return
        self.key_pool[self.key_idx].play()
        self.key_idx = (self.key_idx + 1) % len(self.key_pool)

    def play_click(self):

import sys
import os
import subprocess
from typing import Optional

class WindowInfo:
    def __init__(self, title="", app_name="", window_id=0, pid=0):
        self.title = title
        self.app_name = app_name
        self.window_id = window_id
        self.pid = pid

    def __bool__(self):
        return bool(self.title or self.app_name)

    def format_slot_string(self) -> str:
        clean_app = (self.app_name or "").strip()
        clean_title = (self.title or "").strip()
        if clean_app and clean_title:
            return f"[{clean_app}] {clean_title}"
        elif clean_app:
            return f"[{clean_app}]"
        return clean_title

def get_window_at_point(x: int, y: int) -> WindowInfo:
    """마우스 커서 좌표 (x, y) 아래에 있는 최상위 윈도우 정보를 반환 (게임/보안창 보완용)"""
    if sys.platform == 'win32':
        try:
            import ctypes
            from ctypes import wintypes
            user32 = ctypes.windll.user32
            kernel32 = ctypes.windll.kernel32

            pt = wintypes.POINT(int(x), int(y))
            hwnd = user32.WindowFromPoint(pt)
            if not hwnd:
                return WindowInfo()

            root_hwnd = user32.GetAncestor(hwnd, 2)  # GA_ROOT
            if root_hwnd:
                hwnd = root_hwnd

            win_pid = ctypes.c_ulong()
            user32.GetWindowThreadProcessId(hwnd, ctypes.byref(win_pid))
            pid_val = win_pid.value

            length = user32.GetWindowTextLengthW(hwnd)
            title = ""
            if length > 0:
                buff = ctypes.create_unicode_buffer(length + 1)
                user32.GetWindowTextW(hwnd, buff, length + 1)
                title = buff.value.strip()

            app_name = _get_win32_process_name(kernel32, pid_val)
            return WindowInfo(title=title, app_name=app_name, window_id=hwnd, pid=pid_val)
        except Exception:
            return WindowInfo()

    elif sys.platform == 'darwin':
        try:
            from Quartz import (
                CGWindowListCopyWindowInfo,
                kCGWindowListOptionOnScreenOnly,
                kCGWindowListExcludeDesktopElements,
                kCGNullWindowID,
            )
            options = kCGWindowListOptionOnScreenOnly | kCGWindowListExcludeDesktopElements
            win_list = CGWindowListCopyWindowInfo(options, kCGNullWindowID) or []
            for w in win_list:
                if int(w.get("kCGWindowLayer", 0)) != 0:
                    continue
                bounds = w.get("kCGWindowBounds", {})
                wx, wy = float(bounds.get("X", 0)), float(bounds.get("Y", 0))
                ww, wh = float(bounds.get("Width", 0)), float(bounds.get("Height", 0))
                if wx <= x <= wx + ww and wy <= y <= wy + wh:
                    pid = int(w.get("kCGWindowOwnerPID", 0))
                    app_name = str(w.get("kCGWindowOwnerName") or "").strip()
                    title = str(w.get("kCGWindowName") or "").strip()
                    w_id = int(w.get("kCGWindowNumber", 0)) or pid
                    if app_name or title:
                        return WindowInfo(title=title, app_name=app_name, window_id=w_id, pid=pid)
        except Exception:
            pass

    return WindowInfo()

def _get_win32_process_name(kernel32, pid_val: int) -> str:
    PROCESS_QUERY_LIMITED_INFORMATION = 0x1000
    h_process = kernel32.OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, False, pid_val)
    if not h_process:
        return ""
    try:
        import ctypes
        name_buf = ctypes.create_unicode_buffer(1024)
        size = ctypes.c_ulong(1024)
        if kernel32.QueryFullProcessImageNameW(h_process, 0, name_buf, ctypes.byref(size)):
            return os.path.basename(name_buf.value)
    finally:
        kernel32.CloseHandle(h_process)
    return ""

def _get_darwin_window_quartz() -> WindowInfo:
    try:
        from AppKit import NSWorkspace
        from Quartz import (
            CGWindowListCopyWindowInfo,
            kCGWindowListOptionOnScreenOnly,
            kCGWindowListExcludeDesktopElements,
            kCGNullWindowID,
        )

        front_app = NSWorkspace.sharedWorkspace().frontmostApplication()
        if not front_app:
            return WindowInfo()

        pid = int(front_app.processIdentifier())
        app_name = str(front_app.localizedName() or "")

        options = kCGWindowListOptionOnScreenOnly | kCGWindowListExcludeDesktopElements
        win_list = CGWindowListCopyWindowInfo(options, kCGNullWindowID) or []

        for w in win_list:
            if int(w.get("kCGWindowOwnerPID", 0)) != pid:
                continue
            if int(w.get("kCGWindowLayer", 0)) != 0:
                continue
            w_title = str(w.get("kCGWindowName") or "").strip()
            w_id = int(w.get("kCGWindowNumber", 0)) or pid
            return WindowInfo(title=w_title, app_name=app_name, window_id=w_id, pid=pid)

        return WindowInfo(title="", app_name=app_name, window_id=pid, pid=pid)
    except Exception:
        return WindowInfo()

def _get_darwin_window_applescript() -> WindowInfo:
    try:
        script = '''
        set frontAppName to ""
        set wTitle to ""
        set wId to 0
        set pId to 0
        tell application "System Events"
            try
                set frontApp to first application process whose frontmost is true
                set frontAppName to name of frontApp
                set pId to unix id of frontApp
                try
                    tell frontApp
                        if (count of windows) > 0 then
                            set wTitle to name of front window
                            try
                                set wId to id of front window
                            on error
                                set wId to pId
                            end try
                        else
                            set wId to pId
                        end if
                    end tell
                on error
                    set wId to pId
                end try
            end try
        end tell
        return frontAppName & "|||" & wTitle & "|||" & (wId as string) & "|||" & (pId as string)
        '''
        out = subprocess.check_output(
            ["osascript", "-e", script],
            stderr=subprocess.DEVNULL,
            timeout=0.4
        )
        raw = out.decode("utf-8", errors="ignore").strip()
        parts = raw.split("|||")

        app_name = parts[0].strip() if len(parts) > 0 else ""
        title = parts[1].strip() if len(parts) > 1 else ""
        window_id = int(parts[2]) if len(parts) > 2 and parts[2].strip().isdigit() else 0
        pid = int(parts[3]) if len(parts) > 3 and parts[3].strip().isdigit() else 0
        if window_id == 0 and pid != 0:
            window_id = pid

        return WindowInfo(title=title, app_name=app_name, window_id=window_id, pid=pid)
    except Exception:
        return WindowInfo()

def get_active_window_info() -> WindowInfo:
    if sys.platform == 'win32':
        try:
            import ctypes
            user32 = ctypes.windll.user32
            kernel32 = ctypes.windll.kernel32

            hwnd = user32.GetForegroundWindow()
            if not hwnd:
                return WindowInfo()

            root_hwnd = user32.GetAncestor(hwnd, 2)
            if root_hwnd:
                hwnd = root_hwnd

            win_pid = ctypes.c_ulong()
            user32.GetWindowThreadProcessId(hwnd, ctypes.byref(win_pid))
            pid_val = win_pid.value

            length = user32.GetWindowTextLengthW(hwnd)
            title = ""
            if length > 0:
                buff = ctypes.create_unicode_buffer(length + 1)
                user32.GetWindowTextW(hwnd, buff, length + 1)
                title = buff.value.strip()

            app_name = _get_win32_process_name(kernel32, pid_val)

            if app_name.lower() == "applicationframehost.exe":
                child_pids = []
                WNDENUMPROC = ctypes.WINFUNCTYPE(ctypes.c_bool, ctypes.c_void_p, ctypes.c_void_p)
                def enum_child_proc(c_hwnd, lparam):
                    c_pid = ctypes.c_ulong()
                    user32.GetWindowThreadProcessId(c_hwnd, ctypes.byref(c_pid))
                    if c_pid.value and c_pid.value != pid_val:
                        child_pids.append(c_pid.value)
                        return False
                    return True
                user32.EnumChildWindows(hwnd, WNDENUMPROC(enum_child_proc), 0)
                if child_pids:
                    real_app = _get_win32_process_name(kernel32, child_pids[0])
                    if real_app:
                        app_name = real_app
                        pid_val = child_pids[0]

            return WindowInfo(title=title, app_name=app_name, window_id=hwnd, pid=pid_val)
        except Exception:
            return WindowInfo()

    elif sys.platform == 'darwin':
        info = _get_darwin_window_quartz()
        if info and info.title:
            return info
        # Quartz에서 창 제목이 비어 있는 경우(화면 기록 권한 미허용 등) AppleScript로 보조 조회
        as_info = _get_darwin_window_applescript()
        return as_info if as_info else info

    return WindowInfo()

def is_left_mouse_pressed() -> bool:
    """전역 훅 없이 현재 마우스 왼쪽 버튼이 물리적으로 눌려 있는지 확인"""
    if sys.platform == 'win32':
        try:
            import ctypes
            VK_LBUTTON = 0x01
            return bool(ctypes.windll.user32.GetAsyncKeyState(VK_LBUTTON) & 0x8000)
        except Exception:
            return False
    elif sys.platform == 'darwin':
        try:
            from Quartz import CGEventSourceButtonState, kCGEventSourceStateCombinedSessionState, kCGMouseButtonLeft
            return bool(CGEventSourceButtonState(kCGEventSourceStateCombinedSessionState, kCGMouseButtonLeft))
        except Exception:
            return False
    return False

    _was_mouse_down = False

    capture_bridge.window_captured.emit(slot_str, is_cancelled)


def _is_valid_target_window(win, my_pid: int) -> bool:
    if not win or win.pid == my_pid:
        return False
    title_l = (win.title or "").lower()
    app_l = (win.app_name or "").lower()
    if "morningmeal" in title_l or "morningmeal" in app_l:
        return False
    if app_l == "explorer.exe" and title_l in ("", "program manager"):
        return False
    return bool(win.format_slot_string())


def _poll_window_capture():
    global _capture_elapsed_ms, _was_mouse_down
    _capture_elapsed_ms += 50

    if _capture_elapsed_ms >= 15000:
        _finish_capture("", True)
        return

    my_pid = os.getpid()

    # 1) 포그라운드 창이 다른 프로그램(VS Code, 게임 등)으로 바뀐 경우 즉시 감지
    fg_win = get_active_window_info()
    if _is_valid_target_window(fg_win, my_pid) and fg_win.window_id != _initial_window_id:
        _finish_capture(fg_win.format_slot_string(), False)
        return

    # 2) 마우스 왼쪽 버튼이 눌렸다 떼어지는 순간 커서 아래 창 직접 조회
    is_down = is_left_mouse_pressed()
    if is_down:
        _was_mouse_down = True
    elif _was_mouse_down and not is_down:
        _was_mouse_down = False
        fg_win = get_active_window_info()
        if _is_valid_target_window(fg_win, my_pid):
            _finish_capture(fg_win.format_slot_string(), False)
            return

        pos = QCursor.pos()
        pt_win = get_window_at_point(pos.x(), pos.y())
        if _is_valid_target_window(pt_win, my_pid):
            _finish_capture(pt_win.format_slot_string(), False)
            return


def start_window_click_capture():
    global _capture_poll_timer, _capture_elapsed_ms, _was_mouse_down, _initial_window_id
    cancel_window_click_capture()

    init_win = get_active_window_info()
    _initial_window_id = init_win.window_id if init_win else 0
    _capture_elapsed_ms = 0
    _was_mouse_down = False

    _capture_poll_timer = QTimer()
    _capture_poll_timer.setInterval(50)
    _capture_poll_timer.timeout.connect(_poll_window_capture)
    _capture_poll_timer.start()


def cancel_window_click_capture():
    _finish_capture("", True)

class KeyCaptureButton(QPushButton):
    keyCaptured = pyqtSignal(str)

    def __init__(self, text="", parent=None):
        waiting_text = I18n.tr("input_waiting")
        self.raw_key = text if (text and text != waiting_text) else ""
        super().__init__(self.raw_key or waiting_text, parent)
        self.capturing = False
        I18n.language_changed.connect(self.retranslate_ui)

    def retranslate_ui(self):
        if not self.raw_key:
            self.setText(I18n.tr("input_detecting") if self.capturing else I18n.tr("input_waiting"))

    def _stop_capture_ui(self):
        self.capturing = False
        self.setProperty("activeCapture", False)
        self.setText(self.raw_key if self.raw_key else I18n.tr("input_waiting"))
        self.style().unpolish(self)
        self.style().polish(self)

    def focusOutEvent(self, event: QFocusEvent):
        if self.capturing:
            self._stop_capture_ui()
        super().focusOutEvent(event)

    def mousePressEvent(self, event: QMouseEvent):
        if not self.capturing:
            self.capturing = True
            self.setText(I18n.tr("input_detecting"))
            self.setProperty("activeCapture", True)
            self.style().unpolish(self)
            self.style().polish(self)
            self.setFocus()
            event.accept()
            return

        btn = event.button()
        k_name = "mouse_left" if btn == Qt.MouseButton.LeftButton else ("mouse_right" if btn == Qt.MouseButton.RightButton else "mouse_middle")
        self.raw_key = k_name
        self._stop_capture_ui()
        self.keyCaptured.emit(k_name)
        event.accept()

    def keyPressEvent(self, event: QKeyEvent):
        if not self.capturing:
            super().keyPressEvent(event)
            return

        key = event.key()
        modifiers = event.modifiers()
        if key in (Qt.Key.Key_Shift, Qt.Key.Key_Control, Qt.Key.Key_Alt, Qt.Key.Key_Meta):
            event.accept()
            return

        text = event.text().strip()
        is_shift = bool(modifiers & Qt.KeyboardModifier.ShiftModifier)
        is_ctrl = bool(modifiers & Qt.KeyboardModifier.ControlModifier)
        is_alt = bool(modifiers & Qt.KeyboardModifier.AltModifier)

        key_names = {
            Qt.Key.Key_Space: "space", Qt.Key.Key_Return: "enter", Qt.Key.Key_Enter: "enter",
            Qt.Key.Key_Tab: "tab", Qt.Key.Key_Backspace: "backspace", Qt.Key.Key_Escape: "esc",
            Qt.Key.Key_Left: "left", Qt.Key.Key_Right: "right", Qt.Key.Key_Up: "up", Qt.Key.Key_Down: "down"
        }
        base = key_names.get(key, text.lower() if text else f"key_{key}")

        mod_parts = []
        if modifiers & Qt.KeyboardModifier.MetaModifier:
            mod_parts.append("cmd" if sys.platform == "darwin" else "win")
        if is_ctrl: mod_parts.append("ctrl")
        if is_alt: mod_parts.append("alt")
        if is_shift: mod_parts.append("shift")

        final_key = "+".join(mod_parts) + "+" + base if mod_parts else base
        self.raw_key = final_key
        self._stop_capture_ui()
        self.keyCaptured.emit(final_key)
        event.accept()

class PetWidget(QWidget):
    dock_changed = pyqtSignal()
    scale_changed = pyqtSignal(str, float)
    skin_changed = pyqtSignal(str, str)
    moved_with_timer = pyqtSignal(str, int, int)

    def __init__(self, pet_data, get_all_timers_callback, open_settings_callback, duplicate_callback, remove_callback, quit_callback=None):
        super().__init__()
        self.pet_data = pet_data
        self.get_all_timers_callback = get_all_timers_callback
        self.open_settings_callback = open_settings_callback
        self.duplicate_callback = duplicate_callback
        self.remove_callback = remove_callback
        self.quit_callback = quit_callback

        self.tap_index = 0
        self.scale_x, self.scale_y = 1.0, 1.0
        self.squash_depth = 0.20
        self.skin_enable_bounce = True
        self.stiffness = 0.25
        self.hit_times = deque(maxlen=20)
        self.drag_position = QPoint()
        self._applied_flags = None

        self.setAttribute(Qt.WidgetAttribute.WA_TranslucentBackground, True)
        if sys.platform == 'darwin':
            self.setAttribute(Qt.WidgetAttribute.WA_MacAlwaysShowToolWindow, True)

        self.update_window_flags()

        self.anim_timer = QTimer(self)
        self.anim_timer.timeout.connect(self.update_animation)
        self.reset_timer = QTimer(self)
        self.reset_timer.setSingleShot(True)
        self.reset_timer.timeout.connect(self.reset_to_idle)

        self.load_resources()
        self.move(int(self.pet_data.get("x", 200)), int(self.pet_data.get("y", 200)))
        QTimer.singleShot(100, self.initial_dock_check)

    def initial_dock_check(self):
        t_id = self.pet_data.get("bound_timer_id")
        if t_id:
            for timer_w in self.get_all_timers_callback():
                if timer_w.timer_data.get("id") == t_id:
                    self.snap_to_timer(timer_w)
                    self.save_position()
                    break

    def update_window_flags(self):
        flags = Qt.WindowType.FramelessWindowHint | Qt.WindowType.WindowStaysOnTopHint
        if config_mgr.config.get("settings", {}).get("tray_mode", False):
            if sys.platform == 'win32': flags |= Qt.WindowType.Tool
            else: flags |= Qt.WindowType.NoDropShadowWindowHint
        if config_mgr.config.get("settings", {}).get("click_through", False):
            flags |= Qt.WindowType.WindowTransparentForInput

        if self._applied_flags != flags:
            self._applied_flags = flags
            self.setWindowFlags(flags)
            self.show()

    def load_resources(self):
        skin = self.pet_data.get("skin", "default")
        self.display_scale = float(self.pet_data.get("scale", 1.0))
        skin_conf = config_mgr.get_skin_config(skin)
        self.squash_depth = float(skin_conf.get("squash_depth", 0.20))
        self.skin_enable_bounce = bool(skin_conf.get("enable_bounce", True))
        self.key_mappings = skin_conf.get("key_mappings", {})

        s_dir = os.path.join(SKINS_DIR, skin)
        self.idle_pixmap = QPixmap(os.path.join(s_dir, skin_conf.get("idle_image", "idle.png")))
        self.tap_pixmaps = [QPixmap(os.path.join(s_dir, p)) for p in skin_conf.get("tap_images", ["tap_left.png", "tap_right.png"])]

        self.cached_pixmaps = {}
        for k, v in self.key_mappings.items():
            img_path = os.path.join(s_dir, v)
            if os.path.exists(img_path):
                self.cached_pixmaps[str(k).strip().lower()] = QPixmap(img_path)

        self.current_pixmap = self.idle_pixmap
        self.update_widget_size()
        self.update()

    def update_widget_size(self):
        if not self.idle_pixmap.isNull():
            self.resize(max(30, int(self.idle_pixmap.width() * self.display_scale)), max(30, int(self.idle_pixmap.height() * self.display_scale)))

    def trigger_bounce(self, key_payload):
        bound_id = self.pet_data.get("bound_timer_id", "")
        if bound_id and hasattr(activity_engine, "is_timer_active") and not activity_engine.is_timer_active(bound_id):
            return

        now = time.time()
        self.hit_times.append(now)
        apm = ((len(self.hit_times) / (now - self.hit_times[0])) * 60) if len(self.hit_times) > 1 and (now - self.hit_times[0]) > 0 else 0

        candidates = [c.strip().lower() for c in key_payload.split("|") if c.strip()]
        matched_pixmap = next((self.cached_pixmaps[c] for c in candidates if c in self.cached_pixmaps and not self.cached_pixmaps[c].isNull()), None)

        if matched_pixmap: self.current_pixmap = matched_pixmap
        elif any(c.startswith("mouse_") for c in candidates) and "mouse_click" in self.cached_pixmaps:
            self.current_pixmap = self.cached_pixmaps["mouse_click"]
        else:
            valid_taps = [p for p in self.tap_pixmaps if not p.isNull()]
            if valid_taps:
                self.current_pixmap = valid_taps[self.tap_index % len(valid_taps)]
                self.tap_index = (self.tap_index + 1) % len(valid_taps)

        if self.skin_enable_bounce:
            extra = min(apm / 600.0, 1.0) * 0.08
            depth = min(self.squash_depth + extra, 0.75)
            self.scale_y = max(0.20, 1.0 - depth)
            self.scale_x = 1.0 + (depth * 0.5)
            self.anim_timer.start(16)
        else:
            self.scale_x, self.scale_y = 1.0, 1.0
            if self.anim_timer.isActive(): self.anim_timer.stop()

        self.reset_timer.start(240 if matched_pixmap else 160)
        self.update()

    def update_animation(self):
        if abs(self.scale_x - 1.0) < 0.005 and abs(self.scale_y - 1.0) < 0.005:
            self.scale_x, self.scale_y = 1.0, 1.0
            self.anim_timer.stop()
        else:
            self.scale_x += (1.0 - self.scale_x) * self.stiffness
            self.scale_y += (1.0 - self.scale_y) * self.stiffness
        self.update()

    def reset_to_idle(self):
        self.current_pixmap = self.idle_pixmap
        self.scale_x, self.scale_y = 1.0, 1.0
        self.update()

    def paintEvent(self, event):
        if self.current_pixmap.isNull(): return
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.SmoothPixmapTransform)
        painter.translate(self.width() / 2, self.height())
        painter.scale(self.display_scale * self.scale_x, self.display_scale * self.scale_y)
        painter.translate(-self.current_pixmap.width() / 2, -self.current_pixmap.height())
        painter.drawPixmap(0, 0, self.current_pixmap)

    def wheelEvent(self, event: QWheelEvent):
        if QApplication.keyboardModifiers() == Qt.KeyboardModifier.ControlModifier:
            delta = event.angleDelta().y()
            step = 0.05
            new_scale = round(min(1.50, max(0.10, self.display_scale + (step if delta > 0 else -step))), 2)
            if new_scale != self.display_scale:
                self.display_scale = new_scale
                self.pet_data["scale"] = new_scale
                self.update_widget_size()
                self.initial_dock_check()
                config_mgr.save_config()
                self.update()
                self.scale_changed.emit(self.pet_data.get("id"), self.display_scale)
            event.accept()
        else:
            super().wheelEvent(event)

    def snap_to_timer(self, timer_widget):
        m = getattr(timer_widget, "SHADOW_MARGIN", 0)
        t_x, t_y, t_w = timer_widget.x(), timer_widget.y(), timer_widget.width()
        self.move(int(t_x + (t_w - self.width()) / 2), int(t_y + m - self.height() + 4))

    def bind_to_timer(self, timer_id):
        for p in config_mgr.config.get("pets", []):
            if p.get("bound_timer_id") == timer_id and p.get("id") != self.pet_data.get("id"):
                p["bound_timer_id"] = ""
        self.pet_data["bound_timer_id"] = timer_id
        for tw in self.get_all_timers_callback():
            if tw.timer_data.get("id") == timer_id:
                self.snap_to_timer(tw)
                break
        self.save_position()
        self.dock_changed.emit()

    def unbind_timer(self):
        self.pet_data["bound_timer_id"] = ""
        self.save_position()
        self.dock_changed.emit()

    def save_position(self):
        self.pet_data["x"] = self.x()
        self.pet_data["y"] = self.y()
        config_mgr.save_config()

    def mousePressEvent(self, event: QMouseEvent):
        if event.button() == Qt.MouseButton.LeftButton:
            self.drag_position = event.globalPosition().toPoint() - self.frameGeometry().topLeft()
            event.accept()

    # ui/pet_widget.py -> mouseMoveEvent 교체
    def mouseMoveEvent(self, event: QMouseEvent):
        if event.buttons() == Qt.MouseButton.LeftButton:
            new_pos = event.globalPosition().toPoint() - self.drag_position
            if config_mgr.config.get("settings", {}).get("clamp_to_screen", True):
                screen = QApplication.screenAt(event.globalPosition().toPoint()) or QApplication.primaryScreen()
                if screen:
                    geo = screen.availableGeometry()
                    new_pos = QPoint(max(geo.left(), min(new_pos.x(), geo.right() - self.width())), max(geo.top(), min(new_pos.y(), geo.bottom() - self.height())))
            self.move(new_pos)
            bound_id = self.pet_data.get("bound_timer_id", "")
            if bound_id:
                for tw in self.get_all_timers_callback():
                    if tw.timer_data.get("id") == bound_id:
                        m = getattr(tw, "SHADOW_MARGIN", 0)
                        t_x = int(new_pos.x() + (self.width() - tw.width()) / 2)
                        t_y = int(new_pos.y() + self.height() - m - 4)
                        tw.move(t_x, t_y)
                        tw.timer_data["x"], tw.timer_data["y"] = t_x, t_y
                        break
            event.accept()

    def mouseReleaseEvent(self, event: QMouseEvent):
        self.save_position()
        bound_id = self.pet_data.get("bound_timer_id", "")
        if bound_id:
            for tw in self.get_all_timers_callback():
                if tw.timer_data.get("id") == bound_id:
                    tw.timer_data["x"], tw.timer_data["y"] = tw.x(), tw.y()
                    config_mgr.save_config()
                    break
        else:
            dropped_center = self.geometry().center()
            matched = next((tw for tw in self.get_all_timers_callback() if tw.geometry().adjusted(-20, -70, 20, 20).contains(dropped_center)), None)
            if matched: self.bind_to_timer(matched.timer_data.get("id"))
        event.accept()

    def contextMenuEvent(self, event: QContextMenuEvent):
            menu = StyledMenu(self)
            settings_act = menu.addAction(I18n.tr("open_settings"))
            duplicate_act = menu.addAction(I18n.tr("duplicate_pet"))
            remove_act = menu.addAction(I18n.tr("remove_pet"))
            menu.addSeparator()

            if self.pet_data.get("bound_timer_id", ""):
                menu.addAction(I18n.tr("unbind_timer")).triggered.connect(self.unbind_timer)
            else:
                dock_menu = menu.addStyledMenu(I18n.tr("bind_timer"))
                timers = self.get_all_timers_callback()
                if timers:
                    for tw in timers:
                        t_id = tw.timer_data.get("id")
                        dock_menu.addAction(
                            tw.timer_data.get("name", I18n.tr("default_timer_name"))
                        ).triggered.connect(lambda checked, tid=t_id: self.bind_to_timer(tid))
                else:
                    dock_menu.setEnabled(False)

            menu.addSeparator()
            skin_menu = menu.addStyledMenu(I18n.tr("change_skin"))
            cur_skin = self.pet_data.get("skin", "default")
            if os.path.exists(SKINS_DIR):
                for s_name in sorted(os.listdir(SKINS_DIR)):
                    if os.path.isdir(os.path.join(SKINS_DIR, s_name)):
                        act = skin_menu.addAction(s_name)
                        act.setCheckable(True)
                        if s_name == cur_skin:
                            act.setChecked(True)
                        act.triggered.connect(lambda chk, n=s_name: self.change_skin_direct(n))

            menu.addSeparator()
            exit_act = menu.addAction(I18n.tr("tray_exit"))

            chosen = menu.exec(event.globalPos())
            if chosen == settings_act and self.open_settings_callback:
                self.open_settings_callback()
            elif chosen == duplicate_act and self.duplicate_callback:
                self.duplicate_callback(self.pet_data)
            elif chosen == remove_act and self.remove_callback:
                self.remove_callback(self.pet_data.get("id"))
            elif chosen == exit_act:
                if self.quit_callback:
                    self.quit_callback()
                else:
                    QApplication.quit()

    def change_skin_direct(self, new_skin):
        self.pet_data["skin"] = new_skin
        config_mgr.save_config()
        self.load_resources()
        self.initial_dock_check()
        self.skin_changed.emit(self.pet_data.get("id"), new_skin)
        self.dock_changed.emit()

        skins = [d for d in os.listdir(SKINS_DIR) if os.path.isdir(os.path.join(SKINS_DIR, d))]

            name = t.get("name", I18n.tr("default_timer_name"))

            if t_id == cur_bound: sel_idx = idx + 1

        if not path: return
        skin_name = os.path.splitext(os.path.basename(path))[0].strip()
        target_dir = os.path.join(SKINS_DIR, skin_name)
        try:
            os.makedirs(target_dir, exist_ok=True)
            with zipfile.ZipFile(path, 'r') as zf:
                for member in zf.infolist():
                    if member.filename.startswith("__MACOSX") or member.filename.endswith(".DS_Store"): continue
                    zf.extract(member, target_dir)
            entries = [e for e in os.listdir(target_dir) if e not in ("__MACOSX", ".DS_Store")]
            if len(entries) == 1:
                nested = os.path.join(target_dir, entries[0])
                if os.path.isdir(nested):
                    for item in os.listdir(nested):
                        shutil.move(os.path.join(nested, item), os.path.join(target_dir, item))
                    os.rmdir(nested)
            self.current_editing_skin = skin_name
            self.load_skin_list()
            self.load_current_skin_config()
            self.sync_pet_cards_state()
            AlertDialog(I18n.tr("complete"), I18n.tr("applied"), self).exec()
        except Exception as e:
            QMessageBox.critical(self, I18n.tr("error"), str(e))

        self.combo.clear()
        sounds = scan_sound_files()
        cur = config_mgr.config.get("sound", {}).get(f"{self.prefix}_sound_path", "")
        sel_idx = 0
        idx = 0
        for name, path in sounds.items():
            self.combo.addItem(name, path)
            if cur and (cur == path or os.path.basename(cur) == os.path.basename(path)):
                sel_idx = idx
            idx += 1
        self.combo.addItem(I18n.tr("upload_sound") + "...", "__upload__")
        self.combo.setCurrentIndex(sel_idx)

        if data:
            config_mgr.config.setdefault("sound", {})[f"{self.prefix}_sound_path"] = data
            config_mgr.save_config()
            sound_mgr.load_sounds()
            if self.prefix == "key": sound_mgr.play_key()

            self.populate()
            return

            QMessageBox.warning(self, I18n.tr("error"), I18n.tr("upload_err"))
            self.populate()
            return
        dest = os.path.join(USER_SOUNDS_DIR, os.path.basename(path))
        try:
            if os.path.abspath(path) != os.path.abspath(dest):
                shutil.copy2(path, dest)
            config_mgr.config.setdefault("sound", {})[f"{self.prefix}_sound_path"] = dest
            config_mgr.save_config()
            sound_mgr.load_sounds()
            self.populate()
            if self.prefix == "key": sound_mgr.play_key()
            else: sound_mgr.play_click()
        except Exception as e:
            QMessageBox.critical(self, I18n.tr("error"), str(e))

    def on_group_rename_finished(self):
        new_name = self.name_edit.text().strip()
        if new_name and new_name != self.group_name:
            self.rename_group_callback(self.group_name, new_name)
            self.group_name = new_name

    def retranslate_ui(self):
        self.del_btn.setText(I18n.tr("del_group"))
        for i, lbl in enumerate(self.slot_labels):
            lbl.setText(f"{i+1}{I18n.tr('slot')}")
        for le in self.slot_edits:
            le.setPlaceholderText(I18n.tr("slot_placeholder"))
        for btn in self.slot_pick_btns:
            if not getattr(btn, 'is_capturing', False):
                btn.setText(I18n.tr("pick_window"))
            else:
                btn.setText(I18n.tr("waiting_click"))
        for btn in self.slot_clear_btns:
            btn.setText(I18n.tr("clear"))

    def on_text_changed(self, idx, txt):
        self.windows[idx] = txt.strip()
        config_mgr.config.setdefault("groups", {})[self.group_name] = [w for w in self.windows if w]
        config_mgr.save_config()

    def clear_slot(self, le, idx):
        le.clear()
        self.on_text_changed(idx, "")

    def start_capture(self, btn, le, idx):
        global _active_capture_session
        if getattr(btn, 'is_capturing', False):
            _clear_active_capture_session()
            cancel_window_click_capture()
            return

        _clear_active_capture_session()
        btn.is_capturing = True
        btn.setText(I18n.tr("waiting_click"))
        btn.setProperty("activeCapture", True)
        btn.style().unpolish(btn)
        btn.style().polish(btn)

        _active_capture_session = (self, btn, le, idx)
        start_window_click_capture()

    def reset_btn(self, btn):
        btn.is_capturing = False
        btn.setText(I18n.tr("pick_window"))
        btn.setProperty("activeCapture", False)
        btn.style().unpolish(btn)
        btn.style().polish(btn)

    def on_window_captured(self, slot_text: str, is_cancelled: bool):
        global _active_capture_session
        if _active_capture_session is None:
            return
        card, btn, le, idx = _active_capture_session
        if card == self:
            _active_capture_session = None
            try:
                if not is_cancelled and slot_text:
                    le.setText(slot_text)
                    self.on_text_changed(idx, slot_text)
                self.reset_btn(btn)
            except RuntimeError:
                pass

        config_mgr.save_config()

    def add_timer(self):
        groups = list(config_mgr.config.get("groups", {}).keys())
        first_g = groups[0] if groups else ""
        new_t = {
            "id": str(uuid.uuid4()),
            "name": f"{I18n.tr('default_timer_name')} {len(config_mgr.config.get('timers', [])) + 1}",
            "group": first_g,
            "idle_timeout": 5,
            "elapsed_seconds": 0,
            "x": 200, "y": 300,
            "paused": False
        }
        config_mgr.config.setdefault("timers", []).append(new_t)
        config_mgr.save_config()
        self.refresh_timer_list()
        self.settings_changed.emit()

    def remove_timer(self, timer_data):
        dlg = ConfirmDialog(I18n.tr("del_timer_title"), I18n.tr("del_timer_msg").format(name=timer_data.get('name')), is_danger=True, parent=self)
        if dlg.exec() == ConfirmDialog.DialogCode.Accepted:
            t_id = timer_data.get("id")
            for p in config_mgr.config.get("pets", []):
                if p.get("bound_timer_id") == t_id:
                    p["bound_timer_id"] = ""
            if timer_data in config_mgr.config.get("timers", []):
                config_mgr.config["timers"].remove(timer_data)
            config_mgr.save_config()
            self.refresh_timer_list()
            self.settings_changed.emit()

    def create_group(self):
        groups = config_mgr.config.setdefault("groups", {})
        idx = 1
        name = f"{I18n.tr('default_group_name')} {idx}"
        while name in groups:
            idx += 1
            name = f"{I18n.tr('default_group_name')} {idx}"

        groups[name] = ["", "", ""]
        config_mgr.save_config()
        self.refresh_group_list()
        self.sync_timer_cards_groups()
        self.settings_changed.emit()

    def rename_group(self, old_name, new_name):
        groups = config_mgr.config.setdefault("groups", {})
        if new_name in groups and new_name != old_name:
            return
        if old_name in groups:
            wins = groups.pop(old_name)
            groups[new_name] = wins
            for t in config_mgr.config.get("timers", []):
                if t.get("group") == old_name:
                    t["group"] = new_name
            config_mgr.save_config()
            self.sync_timer_cards_groups()
            self.settings_changed.emit()

    def remove_group(self, name):
        dlg = ConfirmDialog(I18n.tr("del_group"), I18n.tr("del_group_msg").format(name=name), is_danger=True, parent=self)
        if dlg.exec() != ConfirmDialog.DialogCode.Accepted:
            return
        groups = config_mgr.config.get("groups", {})
        if name in groups:
            del groups[name]
        fallback = list(groups.keys())[0] if groups else ""
        for t in config_mgr.config.get("timers", []):
            if t.get("group") == name:
                t["group"] = fallback
        config_mgr.save_config()
        self.refresh_group_list()
        self.sync_timer_cards_groups()
        self.settings_changed.emit()

    def sync_timer_cards_groups(self):
        for card in self.card_map.values():
            card.refresh_groups()

    def refresh_timer_list(self):
        self.card_map.clear()
        while self.timer_list_layout.count():
            item = self.timer_list_layout.takeAt(0)
            if item.widget():
                item.widget().deleteLater()

        for t in config_mgr.config.get("timers", []):
            row = TimerRowWidget(t, self.remove_timer)
            row.changed.connect(self.settings_changed.emit)
            row.name_updated.connect(self.timer_name_changed.emit)
            self.card_map[t.get("id")] = row
            self.timer_list_layout.addWidget(row)
        self.timer_list_layout.addStretch()

    def refresh_group_list(self):
        _clear_active_capture_session()
        cancel_window_click_capture()
        while self.group_list_layout.count():
            item = self.group_list_layout.takeAt(0)
            if item.widget():
                item.widget().deleteLater()
        for g_name, wins in config_mgr.config.get("groups", {}).items():
            sec = GroupSectionWidget(g_name, wins, self.remove_group, self.rename_group)
            self.group_list_layout.addWidget(sec)
        self.group_list_layout.addStretch()

    def retranslate_ui(self):
        self.sub_tabs.setTabText(0, I18n.tr("subtab_timer_list"))
        self.sub_tabs.setTabText(1, I18n.tr("subtab_window_groups"))
        self.snap_cb.setText(I18n.tr("magnetic_snap"))
        self.btn_add_timer.setText(I18n.tr("add_timer"))
        self.btn_create_g.setText(I18n.tr("create_group"))

        for card in self.card_map.values():
            card.retranslate_ui()
        for i in range(self.group_list_layout.count()):
            w = self.group_list_layout.itemAt(i).widget()
            if isinstance(w, GroupSectionWidget):
                w.retranslate_ui()

        pass

# ui/timer_widget.py
import sys
from PyQt6.QtWidgets import QWidget, QLabel, QVBoxLayout, QApplication
from PyQt6.QtCore import Qt, QPoint, QRectF, pyqtSignal
from PyQt6.QtGui import QMouseEvent, QPainter, QColor
from core.config_manager import config_mgr
from core.snap_manager import SnapManager
from core.i18n import I18n
from core.activity_engine import activity_engine
from ui.components import format_time, StyledMenu, ConfirmDialog


class TimerWidget(QWidget):
    SHADOW_MARGIN = 8
    moved = pyqtSignal(str, int, int)
    settings_requested = pyqtSignal()

    def __init__(
        self,
        timer_data,
        get_all_timers_callback,
        get_bound_pet_callback,
        open_main_settings_callback,
        remove_callback=None,
        quit_callback=None,
    ):
        super().__init__()
        self.timer_data = timer_data
        self.get_all_timers_callback = get_all_timers_callback
        self.get_bound_pet_callback = get_bound_pet_callback
        self.open_main_settings_callback = open_main_settings_callback
        self.remove_callback = remove_callback
        self.quit_callback = quit_callback
        self.drag_position = QPoint()
        self._applied_flags = None

        self.setFixedWidth(140 + self.SHADOW_MARGIN * 2)
        self.setAttribute(Qt.WidgetAttribute.WA_TranslucentBackground, True)
        if sys.platform == 'darwin':
            self.setAttribute(Qt.WidgetAttribute.WA_MacAlwaysShowToolWindow, True)

        self.update_window_flags()
        self.init_ui()
        self.move(int(self.timer_data.get("x", 200)), int(self.timer_data.get("y", 300)))
        I18n.language_changed.connect(self.retranslate_ui)

    def update_window_flags(self):
        flags = Qt.WindowType.FramelessWindowHint | Qt.WindowType.WindowStaysOnTopHint
        if config_mgr.config.get("settings", {}).get("tray_mode", False):
            if sys.platform == 'win32':
                flags |= Qt.WindowType.Tool
            else:
                flags |= Qt.WindowType.NoDropShadowWindowHint

        if self._applied_flags != flags:
            was_visible = self.isVisible()
            self._applied_flags = flags
            self.setWindowFlags(flags)
            if was_visible:
                self.show()

    def init_ui(self):
        m = self.SHADOW_MARGIN
        layout = QVBoxLayout(self)
        layout.setContentsMargins(14 + m, 10 + m, 14 + m, 10 + m)
        layout.setSpacing(4)
        layout.setAlignment(Qt.AlignmentFlag.AlignCenter)

        self.label_name = QLabel(self.timer_data.get("name", I18n.tr("default_timer_name")))
        self.label_name.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.label_name.setStyleSheet("background: transparent; font-size: 11px; font-weight: 600; color: #6B7280;")

        self.label_time = QLabel(format_time(self.timer_data.get("elapsed_seconds", 0)))
        self.label_time.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.label_time.setStyleSheet("background: transparent; font-size: 16px; font-weight: bold; font-family: monospace;")

        layout.addWidget(self.label_name)
        layout.addWidget(self.label_time)
        self.apply_theme()

    def paintEvent(self, event):
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)
        painter.setPen(Qt.PenStyle.NoPen)

        is_dark = config_mgr.config.get("settings", {}).get("dark_mode", False)
        m = self.SHADOW_MARGIN
        card_rect = QRectF(m, m, self.width() - m * 2, self.height() - m * 2)

        # 옅은 다중 레이어 드롭 섀도우 (아래쪽으로 살짝 치우친 자연스러운 그림자)
        for i in range(m, 0, -1):
            alpha = int((1.0 - (i / (m + 1))) ** 2 * (26 if is_dark else 16))
            painter.setBrush(QColor(0, 0, 0, alpha))
            shadow_rect = card_rect.adjusted(-i * 0.6, -i * 0.4 + 1.5, i * 0.6, i * 0.8 + 1.5)
            painter.drawRoundedRect(shadow_rect, 6 + i * 0.4, 6 + i * 0.4)

        # 본체 카드
        bg_color = QColor(24, 24, 27, 248) if is_dark else QColor(255, 255, 255, 248)
        painter.setBrush(bg_color)
        painter.drawRoundedRect(card_rect, 6, 6)

    def apply_theme(self):
        is_dark = config_mgr.config.get("settings", {}).get("dark_mode", False)
        is_paused = self.timer_data.get("paused", False)
        name_color = "#A1A1AA" if is_dark else "#6B7280"
        time_color = "#71717A" if is_paused else ("#FFFFFF" if is_dark else "#000000")

        self.label_name.setStyleSheet(f"background: transparent; font-size: 11px; font-weight: 600; color: {name_color};")
        self.label_time.setStyleSheet(f"background: transparent; font-size: 16px; font-weight: bold; font-family: monospace; color: {time_color};")
        self.update()

    def update_time(self, seconds):
        self.timer_data["elapsed_seconds"] = seconds
        self.label_time.setText(format_time(seconds))

    def update_metadata(self):
        self.label_name.setText(self.timer_data.get("name", I18n.tr("default_timer_name")))
        self.label_time.setText(format_time(self.timer_data.get("elapsed_seconds", 0)))
        self.apply_theme()

    def retranslate_ui(self):
        if not self.timer_data.get("name"):
            self.label_name.setText(I18n.tr("default_timer_name"))

    def mousePressEvent(self, event: QMouseEvent):
        if event.button() == Qt.MouseButton.LeftButton:
            self.drag_position = event.globalPosition().toPoint() - self.frameGeometry().topLeft()
            event.accept()
        elif event.button() == Qt.MouseButton.RightButton:
            self.show_context_menu(event.globalPosition().toPoint())
            event.accept()

    def show_context_menu(self, global_pos):
        menu = StyledMenu(self)

        is_paused = self.timer_data.get("paused", False)
        pause_act = menu.addAction(I18n.tr("resume") if is_paused else I18n.tr("pause"))
        pause_act.triggered.connect(self.toggle_pause)

        reset_act = menu.addAction(I18n.tr("reset"))
        reset_act.triggered.connect(self.reset_time)

        delete_act = menu.addAction(I18n.tr("delete"))
        delete_act.triggered.connect(self.delete_timer)

        menu.addSeparator()

        group_title = I18n.tr("group_label").rstrip(":").strip()
        group_menu = menu.addStyledMenu(group_title)
        groups = list(config_mgr.config.get("groups", {}).keys())
        cur_group = self.timer_data.get("group", "")

        if groups:
            for g_name in groups:
                act = group_menu.addAction(g_name)
                act.setCheckable(True)
                if g_name == cur_group:
                    act.setChecked(True)
                act.triggered.connect(lambda checked, name=g_name: self.change_group(name))
        else:
            empty_act = group_menu.addAction(I18n.tr("no_group_available"))
            empty_act.setEnabled(False)

        menu.addSeparator()

        settings_act = menu.addAction(I18n.tr("open_settings"))
        settings_act.triggered.connect(self.open_main_settings)

        menu.addSeparator()

        exit_act = menu.addAction(I18n.tr("tray_exit"))
        exit_act.triggered.connect(self.quit_program)

        menu.exec(global_pos)

    def toggle_pause(self):
        self.timer_data["paused"] = not self.timer_data.get("paused", False)
        config_mgr.save_config()
        self.update_metadata()

    def reset_time(self):
        dlg = ConfirmDialog(
            I18n.tr("reset_confirm_title"),
            I18n.tr("reset_confirm_msg"),
            is_danger=False,
            parent=None,
        )
        if dlg.exec() == ConfirmDialog.DialogCode.Accepted:
            t_id = self.timer_data.get("id")
            self.timer_data["elapsed_seconds"] = 0
            config_mgr.save_config()
            self.update_metadata()
            activity_engine.tick.emit({t_id: 0})

    def delete_timer(self):
        timer_name = self.timer_data.get("name", I18n.tr("default_timer_name"))
        dlg = ConfirmDialog(
            I18n.tr("del_timer_title"),
            I18n.tr("del_timer_msg").format(name=timer_name),
            is_danger=True,
            parent=None,
        )
        if dlg.exec() == ConfirmDialog.DialogCode.Accepted:
            if self.remove_callback:
                self.remove_callback(self.timer_data.get("id"))

    def change_group(self, group_name: str):
        self.timer_data["group"] = group_name
        config_mgr.save_config()
        self.update_metadata()

    def open_main_settings(self):
        if self.open_main_settings_callback:
            self.open_main_settings_callback()

    def quit_program(self):
        if self.quit_callback:
            self.quit_callback()
        else:
            QApplication.quit()

    def mouseMoveEvent(self, event: QMouseEvent):
        if event.buttons() == Qt.MouseButton.LeftButton:
            raw_pos = event.globalPosition().toPoint() - self.drag_position
            all_timers = self.get_all_timers_callback()
            snapped_pos = SnapManager.calculate_snap(self, all_timers, raw_pos)
            self.move(snapped_pos)

            bound_pet = self.get_bound_pet_callback(self.timer_data.get("id"))
            if bound_pet:
                bound_pet.snap_to_timer(self)
            self.moved.emit(self.timer_data.get("id"), snapped_pos.x(), snapped_pos.y())
            event.accept()

    def mouseReleaseEvent(self, event: QMouseEvent):
        self.timer_data["x"] = self.x()
        self.timer_data["y"] = self.y()
        config_mgr.save_config()
        bound_pet = self.get_bound_pet_callback(self.timer_data.get("id"))
        if bound_pet:
            bound_pet.save_position()
        event.accept()
